import hashlib
import json
from html.parser import HTMLParser
from pathlib import Path
from urllib.parse import unquote, urlsplit


ROOT = Path(__file__).resolve().parents[1]
DIST = ROOT / "dist"


class Links(HTMLParser):
    def __init__(self):
        super().__init__()
        self.targets = []

    def handle_starttag(self, tag, attrs):
        if tag not in {"a", "img", "script", "link", "iframe"}:
            return
        self.targets.extend(value for name, value in attrs if name in {"href", "src"} and value)


def main():
    if not (DIST / "index.html").exists():
        raise SystemExit("Build the book first")
    failures = []
    html_files = list(DIST.rglob("*.html"))
    for page in html_files:
        parser = Links()
        parser.feed(page.read_text())
        for target in parser.targets:
            parsed = urlsplit(target)
            if parsed.scheme or parsed.netloc or not parsed.path:
                continue
            path = unquote(parsed.path)
            candidate = DIST / path.lstrip("/") if path.startswith("/") else page.parent / path
            if not candidate.resolve().is_relative_to(DIST.resolve()) or not candidate.exists():
                failures.append(f"Broken link: {page.relative_to(DIST)} -> {target}")
    publication_pdfs = {
        DIST / "research/publications/2026-11-11-pilot-draft" / name
        for name in ("paper.pdf", "slides.pdf")
    }
    publication_pdfs.add(
        DIST / "research/publications/2026-11-11-research-briefing/research-briefing.pdf"
    )
    forbidden = [path for path in DIST.rglob("*") if path.is_file() and (
        path.suffix in {".rs", ".py", ".toml"}
        or (path.suffix == ".pdf" and path not in publication_pdfs)
        or "originals" in path.parts or ".git" in path.parts
    )]
    failures.extend(f"Unexpected private/source artifact in site: {path}" for path in forbidden)
    manifest = json.loads((ROOT / "research/evidence/manifest.json").read_text())
    verified = 0
    absent = 0
    for record in manifest["records"]:
        if record["status"] != "downloaded":
            continue
        path = ROOT / "research/evidence" / record["path"]
        if not path.exists():
            absent += 1
            continue
        data = path.read_bytes()
        if len(data) != record["bytes"] or hashlib.sha256(data).hexdigest() != record["sha256"]:
            failures.append(f"Archive checksum mismatch: {record['path']}")
        verified += 1
    if failures:
        raise SystemExit("\n".join(failures))
    print(f"PASS: {len(html_files)} HTML pages; local links; artifact isolation; {verified} archive checksums")
    if absent:
        print(f"NOTE: {absent} local-only originals absent; reference-only checkout is supported")


if __name__ == "__main__":
    main()

def validate_catalog(actual, catalog):
    groups = catalog["groups"]
    identifiers = [group["id"] for group in groups]
    if len(identifiers) != len(set(identifiers)) or any(
        not re.fullmatch(r"[a-z][a-z0-9-]*", identifier) for identifier in identifiers
    ):
        raise SystemExit("Invalid or duplicate research topic identifiers")
    registered = [name for group in groups for name in group["documents"]]
    if len(registered) != len(set(registered)):
        raise SystemExit("Duplicate primary assignment in research/catalog.json")
    if set(registered) != actual:
        raise SystemExit(f"Research catalog mismatch: unassigned={sorted(actual - set(registered))}; missing={sorted(set(registered) - actual)}")
