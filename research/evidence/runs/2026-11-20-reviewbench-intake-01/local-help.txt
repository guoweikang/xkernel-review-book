usage: local_run.py [-h] [--set {test,full}] [--manifest MANIFEST] [--pr PR]
                    [--limit LIMIT] [--work WORK] [--out OUT]
                    [--pr-agent PR_AGENT] [--producer PRODUCER]
                    [--mirror-org MIRROR_ORG] [--timeout TIMEOUT]
                    [--seed-dir SEED_DIR] [--no-seed] [--skip-existing]

options:
  -h, --help            show this help message and exit
  --set {test,full}
  --manifest MANIFEST
  --pr PR               run only these PR indices
  --limit LIMIT
  --work WORK
  --out OUT
  --pr-agent PR_AGENT   pr-agent executable
  --producer PRODUCER
  --mirror-org MIRROR_ORG
  --timeout TIMEOUT     seconds per PR for pr-agent
  --seed-dir SEED_DIR   mirror checkouts here for `npm run judge -- --repo-dir
                        ...`
  --no-seed             skip populating --seed-dir
  --skip-existing
